//------------------------ // RENDER SPELLCASTING function renderSpellcasting(creature) { const spellcontainer = $("spellcasting-container"); spellcontainer.innerHTML = ""; const blocks = creature.spellcasting || []; if (!blocks.length) { spellcontainer.innerHTML = "

No spellcasting.

"; spellcontainer.classList.add("creatureBlock"); return; } blocks.forEach((block) => { const wrapper = el("div", "BlockItem"); // Title const title = el("div", "SpellcastingTitle", block.name + "."); wrapper.appendChild(title); // Header entries if (block.headerEntries) { const header = el( "div", "SpellcastingHeader", block.headerEntries.map((e) => `

${datacleanse(e)}

`).join("") ); wrapper.appendChild(header); } // At-will spells if (block.will) { const atWill = el( "div", "SpellcastingLine", `${checkheader("atWill")}` + block.will.map((s) => datacleanse(s)).join(", ") ); wrapper.appendChild(atWill); } // Daily spells if (block.daily) { Object.entries(block.daily).forEach(([key, spells]) => { const headerKey = `daily${key}`; const dailyDiv = el( "div", "SpellcastingLine", `${checkheader(headerKey)}` + spells.map((s) => datacleanse(s)).join(", ") ); wrapper.appendChild(dailyDiv); }); } // Leveled spells if (block.spells) { Object.entries(block.spells).forEach(([level, data]) => { let html; if (level === "0") { html = `${checkheader("cantrip")}` + data.spells.map((s) => datacleanse(s)).join(", "); } else { const headerKey = `lvl${level}slots`; html = `${checkheader(headerKey)}${data.slots} slots): ` + data.spells.map((s) => datacleanse(s)).join(", "); } const levelDiv = el("div", "SpellcastingLine", html); wrapper.appendChild(levelDiv); }); } spellcontainer.appendChild(wrapper); }); }
